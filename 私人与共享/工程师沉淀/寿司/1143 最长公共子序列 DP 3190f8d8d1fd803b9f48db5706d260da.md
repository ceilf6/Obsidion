# 1143 最长公共子序列 DP

DP本质就是对 DFS 表格化

```tsx
var longestCommonSubsequence = function(text1, text2) {
    const f = Array.from({length: text1.length+1}, () => new Array(text2.length+1).fill(0))
    // DP本质就是对 DFS 表格化
    // f 的索引对应 dfs 函数的参数
    // 顺序相反，从dfs的逆推 => f正推 从左上推到右下
    for(let i=0;i<text1.length;i++){
        for(let j=0;j<text2.length;j++){
            if(text1[i]===text2[j]){
                f[i+1][j+1]=f[i][j]+1
            }else{
                f[i+1][j+1]=Math.max(f[i][j+1],f[i+1][j])
            }
        }
    }
    return f[text1.length][text2.length]
}
```

```jsx
var longestCommonSubsequence = function(text1, text2) {
    // let ans = 0
    const cache = new Map()
    /**
     * dfs返回当前状况目标答案
     */
    function dfs(p1,p2){
        const key = `${p1},${p2}`
        if(cache.has(key)){
            return cache.get(key)
        }
        if(p1<0 || p2<0){
            return 0
        }

        if(text1[p1] === text2[p2]){
            const res = dfs(p1-1,p2-1) + 1
            cache.set(key,res)
            return res
        }
        const res = Math.max(dfs(p1-1,p2),dfs(p1,p2-1))
        cache.set(key,res)
        return res
    }
    return dfs(text1.length-1, text2.length-1)
};
```
