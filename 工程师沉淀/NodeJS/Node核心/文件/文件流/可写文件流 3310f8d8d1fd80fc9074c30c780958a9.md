# 可写文件流

![fs.createWriteStream(path[, options]).png](%E5%8F%AF%E5%86%99%E6%96%87%E4%BB%B6%E6%B5%81/fs.createWriteStream(path_options).png)

写入 highWaterMark 占满之后后面写入会进入 写入队列等待写入，产生背压

注意写入队列的内存空间是有限的，要小心背压问题

通过 drain 干涸事件解决背压问题

try - while 共用cur

```jsx
const fs = require("fs")
const path = require("path")

const testPath = path.resolve(__dirname, "./testFiles/2.txt")

// 继承自 Writeable
const writeStreamObj = fs.createWriteStream(testPath, {
    flags: "w", // write 写入并覆盖 
    // "a" - append 追加
    encoding: "utf-8", // 文本； Buffer 用 null
    // start: , // 开始写的字节位置
    highWaterMark: 3, // 16 * 1024, // 一次最多写入字节数
    autoClose: true,
})

console.log(writeStreamObj)

// writeStreamObj.on(事件名，回调函数)

// .write 写入数据
const res = writeStreamObj.write("入") // "\nceilf6")
console.log(res) // false 在utf-8中 中文占三个字节导致通道highWaterMark被填满
// true：写入通道没有被填满，接下来的数据可以直接写入，无须排队
// false：写入通道目前已被填满，接下来的数据将进入写入队列
//  // 要特别注意背压问题，因为写入队列是内存中的数据，是有限的

let cur = 0
function tryWrite() {
    let flag = true
    while (cur < 1024 && flag) {
        flag = writeStreamObj.write("a")
        cur++
    }
    if (cur === 1024) {
        console.log("结束写入")
        console.log("干涸次数", testDrainCnt)
    }
}
let testDrainCnt = 0
// 通过 drain 干涸事件解决背压问题
writeStreamObj.on("drain", () => {
    testDrainCnt++
    tryWrite()
})
console.log("不阻塞同步操作")
/*
不阻塞同步操作
结束写入
干涸次数 342
*/

// writeStreamObj.end("\n最后写入")
// 立刻标记这个流结束，后续不允许再继续 .write()
```

```tsx
    // 3. 文件流
    console.time("3. 文件流")
    const rs = fs.createReadStream(from)
    const ws = fs.createWriteStream(to(3))
    rs.on("data", chunk => {
        // 读到一部分数据
        const flag = ws.write(chunk)
        if (!flag) {
            // 下一次写入会有背压
            rs.pause() // 先暂停读取，等待写
        }
    })
    ws.on("drain", () => {
        rs.resume() // 干涸了，可以继续读
    })
    rs.on("end", () => {
        console.timeEnd("3. 文件流")
    })
```

    // 实现复制
    // 1. copyFile
    console.time("1. copyFile")
    await fs.promises.copyFile(from, to(1))
    console.timeEnd("1. copyFile")

    // 2. 先 read 再 write(二进制)
    console.time("2. 先 read 再 write")
    const fileContent = fs.readFileSync(from)
    await fs.promises.writeFile(to(2), fileContent)
    console.timeEnd("2. 先 read 再 write")

    for (let i = 1; i < 4; i++)
        await fs.promises.unlink(to(i)) // 删除文件

    // 4. 文件流pipe() 内部自动处理了背压
    console.time("4. 文件流pipe")
    rs.pipe(ws)
    rs.on("end", () => {
        console.timeEnd("4. 文件流pipe")
    })
    /*
        1. copyFile: 2.167ms
        2. 先 read 再 write: 0.721ms
        3. 文件流: 3.176ms
        4. 文件流pipe: 1.912ms
