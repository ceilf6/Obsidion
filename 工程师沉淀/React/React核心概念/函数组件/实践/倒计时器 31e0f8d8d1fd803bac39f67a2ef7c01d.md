# 倒计时器

# aft Study

用 useRef 保存重渲染重新执行函数过程变量，而不是重新声明

**useEffect** 传入的副作用回调函数中 **return的回调函数** 会在副作用函数清理的时候执行

1. 下一次副作用执行之前
2. 组件卸载时

```jsx
import React, { useEffect,useState,useRef} from 'react'

export default function(props){
    const [time,setTime] = useState(props.time)

    // 1
    const timerRef = useRef(null)
    useEffect(() => {
        timerRef.current = setInterval(() => {
            setTime(pre => pre-1)
        },1000)
        // 提前卸载
        return () => clearInterval(timerRef.current)
    },[])
    useEffect(() => {
        if(time<=0){
            // 倒计时结束卸载
            clearInterval(timerRef.current)
        }
    },[time])

    // 2
    useEffect(() => {
        if (time <= 0) return
        const id = setTimeout(() => setTime(time-1), 1000)
        return () => clearTimeout(id)  // 清理
    }, [time])

    return (
        <div>
            {time}
        </div>
    )
}
```

# pre问题

```jsx
import React, { useEffect,useState} from 'react'

function delay(time){
    return new Promise(resolve => {
        setTimeout(() => {
            resolve()
        },time)
    })
}

export default function(props){
    const [time,setTime] = useState(props.time)

    // 1
    let timer // 每次重新渲染都会重新执行 => undefined，无法达到保存定时器的目的
    useEffect(() => {
        timer = setInterval(() => {
            setTime(pre => pre-1)
        },1000)
        // 缺少返回清理函数
    },[])

    useEffect(() => {
        if(time<0)
            clearInterval(timer)
    },[time])

    // 2
    useEffect(async () => { 
        // useEffect回调不能是 async
        // 因为 async返回Promise，React要求 useEffect的返回值只能是清理函数
        if(time>0){
            await delay(1000)
            setTime(time-1)
        }
    },[time])

    return (
        <div>
            {time}
        </div>
    )
}
```
