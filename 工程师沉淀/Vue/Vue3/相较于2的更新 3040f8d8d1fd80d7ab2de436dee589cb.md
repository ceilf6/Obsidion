1. 没有Vue构造函数

```jsx
/*

createApp(App).mount('#app')
```

这是截断式的更新（Vue2的不兼容了）

1. this 指向

```
      2 中 this 直接指向的是当前实例
      3 中 this 指向的是一个 Proxy 代理对象，由代理对象去对接组件实例
```

![image.png](工程师沉淀/Vue/Vue3/相较于2的更新/image.png)

1. 组合式 API
    
    Vue2 的选项式 API 会导致一个功能相关的内容分散在各个配置中
    
    （虽然mixins 能实现内聚，但是存在**混合之间层次扁平化难以互相操作**、执行顺序等问题）
    
    而 Vue3 composition 可以直接进行 函数级别的交互，例如将一个功能组合的内容作为函数参数传入到另一个功能组合中
    
    https://github.com/ceilf6/Lab/commit/e8d98d705ca017baa9b5a4fd2f9621f1f363002c
    

```jsx
<template>
  <button @click="handleClick">
    {{
      count
      /* 在通过实例代理访问 count 的时候，代理对象会自动解包返回 .value */
    }}
  </button>
</template>

<script>
import { ref } from "vue";

export default {
  setup() {
    // 在所有生命周期钩子函数之前调用
    // 并且 setup 的 this -> undefined
    console.log(this);

    // let count = 0;
    let count = ref(0);
    console.log(count); // ref 将变量封装到了对象中，设置了访问器进行依赖收集
    const handleClick = () => {
      console.log("handle click");
      // count++; // 没有 ref 的 count 增加时没有 re-render 、即变量不是响应式的
      count.value++; // 在 setup 中，还没有代理、变量是一个对象
    };
    return {
      count,
      handleClick,
    }; // setup 返回的所有属性都会挂载到实例对象上
  },
};
</script>

```

Ref 变量

1. 在例如模版中通过实例代理访问 countRef 的时候，代理对象会自动解包返回 .value
2. 在 setup() 中，还没有代理、变量是一个对象

![image.png](工程师沉淀/Vue/Vue3/相较于2的更新/image%201.png)

那么就能很轻松的实现内聚抽离

```jsx
import { ref } from "vue";

export default function useCount(defaultCnt = 0) {
    // let countRef = 0;
    let countRef = ref(defaultCnt);
    console.log(countRef); // ref 将变量封装到了对象中，设置了访问器进行依赖收集
    const handleIncrease = () => {
        console.log("handle click");
        // countRef++; // 没有 ref 的 countRef 增加时没有 re-render 、即变量不是响应式的
        countRef.value++; // 在 setup 中，还没有代理、变量是一个对象
    };
    const handleDecrease = () => {
        console.log("handle click");
        // countRef++; // 没有 ref 的 countRef 增加时没有 re-render 、即变量不是响应式的
        countRef.value--; // 在 setup 中，还没有代理、变量是一个对象
    };

    return {
        countRef,
        handleIncrease,
        handleDecrease
    }
}
```
