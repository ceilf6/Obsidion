# 并发执行

# 调度器模式

任务队列 => 调度器 => running<limit ? 执行 : 等待

Promise本质是一个未来才能得到的值的占位符

创建 Promise 后先保存 它的resolve 等到完成后再调用 

```jsx
function concurrenceAct(limitNum){
	this.queue = []
	this.limitNum = limitNum
	this.runningNum = 0
	this.res = []
	this.allResolved = null // 是否全部完成
}

concurrenceAct.prototype.startRun = function(tasks){
	this.res = new Array(tasks.length)

	return new Promise((resolve, reject) => {
		this.allResolved = resolve
		tasks.forEach((task, idx) => {
			// idx 用于维护 res 的次序
			this.add(task, idx)
		})
	})
}

concurrenceAct.prototype.add = function(task, idx){
	// idx 用于维护 res 的次序
	return new Promise(resolve => {
		this.queue.push({
			task,
			idx,
			resolve
		})
		this._schedule()
	})
}

concurrenceAct.prototype._schedule = function(){
	while(this.runningNum < this.limitNum && this.queue.length){
		const { task, idx, resolve } = this.queue.shift()
		this.runningNum++
		Promise.resolve()
			.then(task)
			.then(res => {
				this.res[idx] = res
				resolve(res)
			})
			.catch(e => {
				this.res[idx] = e
				resolve(e) // 不中断
			})
			.finally(() => {
				this.runningNum--
				this._schedule() // 继续尝试跑
				this._checkAllResolved()
			})
	}
}

concurrenceAct.prototype._checkAllResolved = function(){
	if(this.runningNum === 0 && this.queue.length === 0){
		this.allResolved(this.res)
	}
}
```

# 无限循环的worker+闭包共享idx抢任务

```jsx
async function doFetch(task){
	return fetch(task)
}

const concurrenceAct = async function(tasks, limitNum){
	const TN = tasks.length
	const res = new Array(TN)
	
	let idx = 0;
	function getIdx(){
		if(idx >= TN) return null
		return idx++
	}
	
	async function worker(){
		while(true){
			const curIdx = getIdx()
			if(curIdx === null) return
			
			const task = tasks[curIdx]
			try{
				if(Object.prototype.toString.call(task) === '[object Function]'){
					res[curIdx] = await task()
				}else{
					res[curIdx] = await doFetch(task)
				}
			}catch(e){
				// throw e // 不能直接throw否则这个工作者后面就不干活了
				res[curIdx] = e
			}
		}
	}
	
	const workers = Array.from({length: Math.min(TN, limitNum)}, () => worker())
	await Promise.all(workers)
	return res
}
```
