# 76 右拖左双指针 滑动窗口

```jsx
/**
 * @param {string} s
 * @param {string} t
 * @return {string}
 */
var minWindow = function(s, t) {
    let ansStart;
    let ansLen = Infinity
    let todoNum = t.length
    const todoChrs = new Map()
    for(const chr of t){
        todoChrs.set(chr, (todoChrs.get(chr) || 0)+1)
    }
    let l=0
    let r=0
    for(let r=0;r<s.length;r++){
        const curChr = s[r]
        if(todoChrs.has(curChr)){
            if(todoChrs.get(curChr)>0)
                todoNum--
            todoChrs.set(curChr,todoChrs.get(curChr)-1)
        }

        while(todoNum===0){ // 如果覆盖了就尝试不断收缩，拖动左指针
            if(r-l+1 < ansLen){
                ansLen = r-l+1
                ansStart = l
            }
            const leftCh = s[l]
            if(todoChrs.has(leftCh)){
                todoChrs.set(leftCh,todoChrs.get(leftCh)+1)
                if(todoChrs.get(leftCh)>0) // 注意得大于0再加，因为前面-1可能导致小于0
                    todoNum++
            }
            l++ // 拖动左指针
        }
    }
    return ansLen === Infinity ? "" : s.slice(ansStart,ansStart+ansLen)
};
```
