## II

hold 状态切换会导致 当前持有金额的变化

```jsx
var maxProfit = function(prices){
    const LP = prices.length
    const f = Array.from({length: LP+1}, () => [-Infinity,-Infinity])
    f[0][0] = 0
    for(let i=0;i<LP;i++){
        f[i+1][0] = Math.max(f[i][0],f[i][1]+prices[i])
        f[i+1][1] = Math.max(f[i][1],f[i][0]-prices[i])
    }
    return f[LP][0]
}

var maxProfit = function(prices) {
    const LP = prices.length
    const cache = Array.from({length: LP}, () => [Infinity,Infinity])

    function dfs(i,hold){
        if(i<0){
            // 如果 <0 都 hold 肯定不行，直接返回负无穷、不纳入考虑
            return hold ? -Infinity : 0
        }

        if(cache[i][hold]!==Infinity)
            return cache[i][hold]

        let res
        if(hold){
            res = Math.max(dfs(i-1,1), dfs(i-1,0)-prices[i])
        } else {
            res = Math.max(dfs(i-1,0), dfs(i-1,1)+prices[i])
        }
        cache[i][hold]=res
        return res
    }
    return dfs(LP-1,0)
};

/**
 * 贪心解法
 */
var maxProfit = function(prices) {
    let ans = 0;
    const L = prices.length;
    let now = prices[0];
    for(let i =1;i<L;i++){
        if(now<prices[i]){
            ans+=prices[i]-now;
        }
        now=prices[i];
    }
    return ans;
};
```

## IV k次

```jsx
var maxProfit = function(k,prices){
    const LP = prices.length
    const f = Array.from({length: LP+1}, () => 
        Array.from({length: k+2}, () => [-Infinity, -Infinity])
    )
    for(let i=1;i<k+2;i++){
        f[0][i][0] = 0
    }
    for(let i=0;i<LP;i++){
        for(let j=1;j<k+2;j++){
            f[i+1][j][0] = Math.max(f[i][j][0],f[i][j-1][1]+prices[i])
            f[i+1][j][1] = Math.max(f[i][j][1],f[i][j][0]-prices[i])
        }
    }
    return f[LP][k+1][0]
}

var maxProfit = function(k,prices) {
    const LP = prices.length
    const cache = Array.from({ length: LP }, () =>
        Array.from({ length: k + 1 }, () => [undefined, undefined])
    )

    function dfs(i,j,hold){
        if(i<0){
            // 如果 <0 都 hold 肯定不行，直接返回负无穷、不纳入考虑
            return hold ? -Infinity : 0
        }
        if(j<0) // 剩余交易次数小于0，那就是不可能
            return -Infinity

        if(cache[i][j][hold]!==undefined)
            return cache[i][j][hold]

        let res
        if(hold){
            res = Math.max(dfs(i-1,j,1), dfs(i-1,j,0)-prices[i])
        } else {
            // 规定卖出时交易次数-1
            res = Math.max(dfs(i-1,j,0), dfs(i-1,j-1,1)+prices[i])
        }
        cache[i][j][hold]=res
        return res
    }
    return dfs(LP-1,k,0)
};
```
