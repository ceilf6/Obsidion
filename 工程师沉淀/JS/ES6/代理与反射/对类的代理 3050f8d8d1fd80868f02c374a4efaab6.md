# 对类的代理

语法糖

```jsx
// ES6 类语法
class RefImpl {
    get value() {
        return this._value
    }
    set value(newValue) {
        this._value = newValue
    }
}

// 等价于在构造后使用 Object.defineProperty
class RefImpl {
    constructor() {
        Object.defineProperty(this, 'value', {
            get() {
                return this._value
            },
            set(newValue) {
                this._value = newValue
            },
            enumerable: true,
            configurable: true
        })
    }
}
```

export default (Class, ...propNames) => {
    return new Proxy(Class, {
        construct(target, args) {
            const obj = Reflect.construct(target, args) // 第二个参数规定是args数组 
            // 拿到底层API后的结果

            // for (let i = 0; i < propNames.length; i++) {
            //     **Object.defineProperty(obj, propNames[i], {
            //         value: args[i]**, // 
            //         writable: true,
            //         enumerable: true,
            //         configurable: true
            //     })
            // }
            propNames.forEach((name, idx) => {
                obj[name] = args[idx]
            })

            return obj
        }
    })
}
