function main(config) {
  const FRONTIER_NAME = "US-Frontier-SSH";
  const CLAUDE_GROUP = "Claude-住宅出口";

  const FRONTIER_HOST = "108.242.235.178";
  const FRONTIER_PORT = 22;
  const FRONTIER_USER = "root";
  const FRONTIER_PASS = "kZNWNr7dbZq38mbW";

  config.proxies = config.proxies || [];
  config["proxy-groups"] = config["proxy-groups"] || [];
  config.rules = config.rules || [];

  // 清理旧节点 / 旧组 / 旧规则
  config.proxies = config.proxies.filter(
    (p) =>
      p.name !== FRONTIER_NAME &&
      p.name !== "US-Frontier-Residential"
  );

  config["proxy-groups"] = config["proxy-groups"].filter(
    (g) => g.name !== CLAUDE_GROUP && g.name !== "Frontier-前置机场"
  );

  config.rules = config.rules.filter((rule) => !rule.includes(CLAUDE_GROUP));

  // 新增 SSH 出口节点：直接连接 108.242.235.178:22
  config.proxies.push({
    name: FRONTIER_NAME,
    type: "ssh",
    server: FRONTIER_HOST,
    port: FRONTIER_PORT,
    username: FRONTIER_USER,
    password: FRONTIER_PASS,

    // 留空等于接受任意 host key；先跑通可以这样，后续建议固定 host-key
    "host-key": [],
  });

  // Claude 专用组
  config["proxy-groups"].push({
    name: CLAUDE_GROUP,
    type: "select",
    proxies: [FRONTIER_NAME, "DIRECT"],
  });

  config.rules = [
    `DOMAIN-SUFFIX,anthropic.com,${CLAUDE_GROUP}`,
    `DOMAIN-SUFFIX,claude.ai,${CLAUDE_GROUP}`,
    ...config.rules,
  ];

  return config;
}
