对购物车信息对象进行规范化

```tsx
import { MopCartItemInfoVO } from '../types/index';

/**
 * 规范化 MopCartItemInfoVO 数据中的 dealId
 * 由于dealId字段未来会被废弃，一段时间后，后端传过来值为负值/空值；但前端所有逻辑都使用dealId字段。
 * 所以需要将 dealIdStr 转换为 number 类型并赋值给 dealId 字段
 * @param mopCartItemInfoVO - 原始的购物车项信息对象
 * @returns 规范化后的购物车项信息对象
 */
export const normalizeCartItemInfoVO = (mopCartItemInfoVO: MopCartItemInfoVO): MopCartItemInfoVO => {
  // 如果对象不存在，直接返回
  if (!mopCartItemInfoVO) {
    return mopCartItemInfoVO;
  }
  // 获取 mopDealMappingProductInfoVO 中的 dealIdStr
  const dealIdStr = mopCartItemInfoVO?.mopDealMappingInfoVO?.mopDealMappingProductInfoVO?.dealIdStr;

  // 如果 dealIdStr 不存在或为空，直接返回原对象
  if (!dealIdStr) {
    return mopCartItemInfoVO;
  }
  // 将 dealIdStr 转换为 number
  const dealIdNumber = Number(dealIdStr);
  // 校验转换结果是否有效
  if (isNaN(dealIdNumber)) {
    return mopCartItemInfoVO;
  }

  // 返回更新后的对象
  return {
    ...mopCartItemInfoVO,
    mopDealMappingInfoVO: {
      ...mopCartItemInfoVO.mopDealMappingInfoVO,
      mopDealMappingProductInfoVO: {
        ...mopCartItemInfoVO.mopDealMappingInfoVO?.mopDealMappingProductInfoVO,
        dealId: dealIdNumber, // 此时dealId为number类型
        dealIdStr,
      },
    },
  };
};

```

监听 originalMopCartItemInfoVO 数据的变化

```tsx
  // 监听 originalMopCartItemInfoVO 变化，控制弹窗显示和数据更新
  useEffect(() => {
    // 拦截转换，normalizeCartItemInfoVO函数将dealId根据新传入的dealIdStr赋值；具体原因看函数内部注释。
    const normalized = normalizeCartItemInfoVO(originalMopCartItemInfoVO);

    // 关键逻辑：只在有数据时更新 mopCartItemInfoVO，数据清空时不更新（保持旧值）
    if (normalized?.cartItemId || normalized?.purchaseItemType) {
      /* 
      有数据时：从“购物车”拉起“修改”浮层，保存 mopCartItemInfoVO数据&&哪怕浮层关闭，这个数据也锁死（数据不用于展示，只是为了兜底）；
      只有重新从“购物车”拉起“修改”浮层，才更新mopCartItemInfoVO数据；
      */
      setMopCartItemInfoVO(normalized);
      setVisible(true);
    } else {
      /* 
        数据清空时，不更新 mopCartItemInfoVO（保持旧值），只关闭浮层
        原因：visible变为false，ShopCustomization组件只在SlideModal组件滑动动画关闭之后才进行卸载；
        但此时用户仍然能够点击&&调用confirmModifyChoose函数，此时访问 mopCartItemInfoVO(undefined).purchaseItemType → 报错
      */
      setVisible(false);
    }
  }, [originalMopCartItemInfoVO]);
```
