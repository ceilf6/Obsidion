### 1. 开发及调试

### 1.1. 项目创建

在 MRN 中，发布是以 Bundle 为单位的，一个 Bundle 可以包含多个 Component。创建 MRN 项目本质上是创建一个个的 MRN Bundle。
@mrn/mrn-cli 的 init 脚本现已废弃并推荐使用 Talos 的 [MRN 创建脚本生成器](https://talos.sankuai.com/#/cli/MRN)，可以生成一个 MRN 项目创建脚本到本地执行。Talos 生成的创建脚本如下：

```tsx
declare preset='{"plugins":{"@mrn/mrn-cli-plugin-setup":{"version":"^3.0.0","name":"mrnproject","biz":"meishi","bundleType":1,"description":"A-MRN-PROJECT","moduleName":"mrnproject","author":"liyiming14","spaceCount":2,"semicolon":false},"@hfe/common-cli-plugin-cleanup-vue":{}}}';

declare createProject='CLI_NAME="Talos CLI" SCRIPT_NAME=start DOCS_LINK=http://mrn.sankuai.com/ talos create --inlinePreset $preset mrnproject --registry=http://r.npm.sankuai.com';

declare installTalosCli='npm i -g @hfe/talos-cli --registry=http://r.npm.sankuai.com';

if [[ $(which talos) != 'talos not found' ]]; then
  if [[ $(talos --version) > 3.1.0 ]]; then
    eval $createProject
  else
    echo '' && echo 发现旧版 @hfe/talos-cli@$(talos --version)，将升级新版，稍后开始创建
    eval $installTalosCli
    eval $createProject
  fi
else
  echo '' && echo 未检测到 talos 命令，先安装 @hfe/talos-cli，稍后开始创建
  eval $installTalosCli
  eval $createProject
fi
```

生成的项目结构为：
项目关键文件：
- mrn.config.js：Bundle 配置，包含 Bundle 的业务线、bundleName 和 debug 配置等等，参考[mrn.config.js](https://mrn.sankuai.com/docs/guide/conf.html#mrn-config-js)。可以导出对象配置单Bundle，也可以导出数组配置多个Bundle，多 Bundle 时可使用 --entryName 指定具体需要调试、构建的 bundleName，参考[start 开发调试工程](https://mrn.sankuai.com/docs/guide/cli.html#_2%E3%80%81start-%E5%BC%80%E5%8F%91%E8%B0%83%E8%AF%95%E5%B7%A5%E7%A8%8B)。
- index.tsx：Bundle 入口，用来注册组件，可以注册多个组件。
- src：项目源代码，默认采用的是 pages、components这种文件接口，App.tsx为组件入口，引入 routers.ts 进行路由设置。
项目创建完毕之后，package.json 中已经有了平时开发所需要的命令：

```tsx
"build": "mrn build",
"start": "mrn start"
```

分别对应构建和开发，这也是平时最常用的两个命令。

### 1.3.1. CSS 样式相关

### 1.3.1.1. rem 单位

在 MRN 中，标准的设备宽度是 375 rem()，如果我们代码写的宽度是 x rem，设备实际宽度是 W，那最终应用的宽度是 x/375*W px。
推荐使用 `@nib/mrn-base-utils` 包中的 RemStyleSheet.create 函数替代原生的 StyleSheet.create 来样式，宽高等数字会根据设备宽度按比例调整。

```tsx
import { RemStyleSheet } from '@meishi/hammer-mrn';
import { rem } from '@meishi/hammer-mrn'; // rem 函数将 rem 转为具体 px
import { RemStyleSheet } from '@nib/mrn-base-utils'
import { RemStyleSheet } from '@nibfe/merchant-mc-utils'

export const styles = RemStyleSheet.create({
  alignItems: {
    alignItems: 'center'
  },
  wrapperStyle: {
    marginTop: 5,
  }
})
```

推荐使用 iPhone X 调试样式，因为 iPhone X的宽度恰好为 375px(iPhone X的宽度恰好为 375px：iOS Device Display Summary)，检查样式时可以避免转换。下图是 iPhone X 调试样式时的图例：
拿到视觉稿先看视觉稿上的屏幕宽度，例如，如果是 750px，表示其为 2 倍图，对应的宽高都应该除以 2 ，变成屏幕宽度为 375 rem 下的标准值写在代码里。上图就是一个二倍图的 UI 稿。

###### 1.3.1.4. CSS 预处理器（SCSS/Less）

初始化创建的 MRN 项目是不支持 import 'xxx.css' 的语法的，都是使用 RemStyleSheet (@meishi/hammer-mrn)或者 StyleSheet(@mrn/react-native) 套一个 JS 对象()来创建样式，因此需要自定义 mrn-cli 的编译配置添加自定义 loader。
以支持 CSS 和 SCSS 为例，创建 config/metro.config.js 文件并修改 package.json 中的 start 和 build 脚本：
```typescript
module.exports = {
  addMetroHandler: metroConf => {
    // 添加扩展名支持
    metroConf.resolver.sourceExts.push('css')
    metroConf.resolver.sourceExts.push('scss')
    return metroConf
  },
  addTransformer: () => [require.resolve('./transform-css-loader')]
}
```

```tsx
{
  "scripts": {
    "start": "mrn start --metroConfig ./config/metro.config.js",
    "build": "mrn build --metroConfig ./config/metro.config.js"
  }
}
```

还需要自己写一个 transform-css-loader，使用 Babel 对 CSS 及 SCSS 文件进行加载，转换成合适的 JS 代码，我们这里把 CSS 读出来，用 RemStyleSheet 创建好样式对象并导出：

```tsx
const postcss = require('postcss')
const postcssJs = require('postcss-js')
const upstreamTransformer = require('metro-react-native-babel-transformer')
const transformCss = require('./transform')

function transform(src, filename, options) {
  const root = postcss.parse(src)
  const cssObj = postcssJs.objectify(root)
  const result = transformCss(cssObj)
  const content = `
   import { RemStyleSheet } from '@nib/mrn-base-utils';
   export default RemStyleSheet.create(${JSON.stringify(result)})
 `
  return upstreamTransformer.transform({
    src: content,
    filename,
    options
  })
}

module.exports = {
  extname: ['.css', '.scss'],
  transform
}
```

```tsx
const compileCss = require('css-to-react-native').default
const borderTransform = require('./border-transform')

module.exports = function (cssObj) {
  const result = {}
  Object.keys(cssObj).forEach(function (classname) {
    const ruleObj = cssObj[classname]
    const ruleArray = Object.keys(ruleObj).reduce((prev, ruleName) => {
      const ruleDetail = ruleObj[ruleName]
      if (borderTransform.condition(ruleName)) {
        prev.push(...borderTransform.transform(ruleName, ruleDetail))
      } else {
        prev.push([ruleName, ruleDetail])
      }
      return prev
    }, [])
    const rnCss = compileCss(ruleArray)
    classname.split(',').forEach(item => {
      const newClassname = item
        .replace(/^[.#]/, '') // 去掉类名开始的. #
        .replace(/[A-z][A-z\\d]*( +[.#][A-z][A-z\\d]*)+/g, text => text.replace(/ +[.#]/g, '_')) // 将a .b #c转化为a_b_c
      result[newClassname] = { ...result[newClassname], ...rnCss }
    })
  })
  return result
}
```

```tsx
const { getStylesForProperty } = require('css-to-react-native')

function transform(ruleName, ruleDetail) {
  const { borderWidth, borderColor, borderStyle } = getStylesForProperty('border', ruleDetail, true)
  return [
    [`${ruleName}Width`, `${borderWidth}`],
    [`${ruleName}Color`, borderColor],
    ['borderStyle', borderStyle]
  ]
}

function condition(ruleName) {
  return ['borderLeft', 'borderRight', 'borderBottom', 'borderTop'].indexOf(ruleName) !== -1
}

module.exports = {
  transform,
  condition
}
```

若要支持 SCSS，可以参考 [postcss-sass](https://github.com/csstools/postcss-sass)，修改上述 postcss 的使用，加入 postcss-sass 插件和 postcss-scss 语法：

```tsx
const postcss = require('postcss');
const postcssSass = require('@csstools/postcss-sass');

postcss([
  postcssSass(/* pluginOptions */)
]).process(YOUR_CSS, {
  syntax: require('postcss-scss')
});
```

若要支持 Less，同理，使用对应的 Less 插件即可。

const result = NativeModules.MRNPageRouter.openUrlWithResult(url, {

###### 1.3.2.3. @mrn/mrn-knb

KNB由客户端组件Titans和JS-SDK构成，作为美团的基础组件，被相当多其他组件依赖，**几乎所有美团内App**使用的H5都是由KNB Web容器承载的。

Pillow 官网：[<https://pillow.sankuai.com/#/home>](<https://pillow.sankuai.com/#/home>)
我们可以根据自己的需要直接修改自定义 JSON 并扫码调试，下面是一个 JSON 示例，其为一个数组，每个元素表示一个插件及其配置，我注释了各个步骤的意义：
```typescript
[
    {
        "index": 0,
        // Pillow 插件「MRN 锁包」
        "type": "mrnbundle",
        "blockOnFail": true,
        "info": {
            // 测试环境
            "isStage": 1,
            // Bundle 名称
            "bundleName": "rn_meishi_ka-mop-order-detail",
            "bundleVersion": "0.29.5"
        }
    },
    {
        "index": 2,
        // Pillow 插件「页面跳转」
        "type": "scheme",
        "info": {
            // 需要跳转的 scheme 地址，不同 App 不同，可参考：<https://mrn.sankuai.com/docs/guide/mrn-url-guide.html#scheme-host>
            "value": "imeituan://www.meituan.com/mrn?mrn_biz=meishi&mrn_entry=ka-mop-order-detail&mrn_component=kaMopOrderDetail"
        },
        "priority": 998,
        "serial": false
    }
]
```
上面这个配置的含义是：首先锁包 rn_meishi_ka-mop-order-detail 到版本 0.29.5，然后执行 scheme 跳转到 imeituan://www.meituan.com/mrn?mrn_biz=meishi&mrn_entry=ka-mop-order-detail&mrn_component=kaMopOrderDetail，也就是这个 Bundle 中 kaMopOrderDetail 组件。

#### 3.4. 各种 ID 的含义

[UserID & UUID & DPID & UnionID & PushToken](<https://km.sankuai.com/page/144946797>)
uuid 是美团侧唯一设备 id，dpid 是点评侧唯一设备id，unionid 是希望以后两侧可以统一为一个唯一的设备 id，目前与 uuid 和 dpid 处于共存的状态。PushToken 是用来给设备发送推送的唯一标示码，各个 App 上的 PushToken 不尽相同。
获取路径：
	- 美团内测：摇一摇 -> 用户和设备信息 ->  userid -> 点击即可复制成功
